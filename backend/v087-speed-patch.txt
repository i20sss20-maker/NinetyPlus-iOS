// v0.8.7 — keep complete lineup metadata, but never hold the match center for
// sequential 8–10 second coach probes. Probe independent public sources in
// parallel, cache misses briefly, and return the first verified coach found.
function replaceFunctionBlock(startMarker,endMarker,replacement,label) {
  const start=s.indexOf(startMarker);
  const end=start>=0 ? s.indexOf(endMarker,start) : -1;
  if (start<0 || end<0) throw new Error(`v0.8.7 patch missing: ${label}`);
  s=s.slice(0,start)+replacement+'\n'+s.slice(end);
}
